 #config LARGUISIMA (bashrc :v)
  programs.bash = {
    shellAliases = {
      ls = "ls --color=auto";
      grep = "grep --color=auto";
      fgrep = "fgrep --color=auto";
      egrep = "egrep --color=auto";
      diff = "diff --color=auto";
      ip = "ip --color=auto";
    };

    interactiveShellInit = ''
      # Ignorar duplicados y comandos que empiezan con espacio en el historial
      HISTCONTROL=ignoreboth
      shopt -s histappend
      HISTSIZE=1000
      HISTFILESIZE=2000

      # Ajustar tamaño de ventana
      shopt -s checkwinsize

      # Detección de color en la terminal
      case "$TERM" in
          xterm-color|*-256color) color_prompt=yes;;
      esac
      force_color_prompt=yes

      if [ -n "$force_color_prompt" ]; then
          if command -v tput >/dev/null 2>&1 && tput setaf 1 >&/dev/null; then
              color_prompt=yes
          else
              color_prompt=
          fi
      fi

      # Configuración del Prompt (Estilo Kali de 2 líneas)
      PROMPT_ALTERNATIVE=twoline
      NEWLINE_BEFORE_PROMPT=yes

      if [ "$color_prompt" = yes ]; then
          VIRTUAL_ENV_DISABLE_PROMPT=1

          prompt_color="\[\033[0m\]"
          info_color="\[\033[0m\]"
          prompt_symbol="@"

          case "$PROMPT_ALTERNATIVE" in
              twoline)
                  PS1="''${prompt_color}┌──''${debian_chroot:+(''${debian_chroot})──}''${VIRTUAL_ENV:+(\[\033[0;1m\]\$(basename ''$VIRTUAL_ENV)''${prompt_color})}(''${info_color}\u''${prompt_symbol}\h''${prompt_color})-[\[\033[0;1m\]\w''${prompt_color}]\n''${prompt_color}└─''${info_color}\$\[\033[0m\] "
                  ;;
              oneline)
                  PS1="''${VIRTUAL_ENV:+(\$(basename ''$VIRTUAL_ENV)) }''${debian_chroot:+(''${debian_chroot})}''${info_color}\u@\h\[\033[00m\]:''${prompt_color}\[\033[01m\]\w\[\033[00m\]\$ "
                  ;;
              backtrack)
                  PS1="''${VIRTUAL_ENV:+(\$(basename ''$VIRTUAL_ENV)) }''${debian_chroot:+(''${debian_chroot})}\[\033[01;31m\]\u@\h\[\033[00m\]:\[\033[01;34m\]\w\[\033[00m\]\$ "
                  ;;
          esac
          unset prompt_color
          unset info_color
          unset prompt_symbol
      else
          PS1="''${debian_chroot:+(''${debian_chroot})}\u@\h:\w\$ "
      fi
      unset color_prompt force_color_prompt

      # Título de la ventana de la terminal
      case "$TERM" in
      xterm*|rxvt*|Eterm|aterm|kterm|gnome*|alacritty)
          PS1="\[\e]0;''${debian_chroot:+(''${debian_chroot})}\u@\h: \w\a\]$PS1"
          ;;
      esac

      [ "$NEWLINE_BEFORE_PROMPT" = yes ] && PROMPT_COMMAND="PROMPT_COMMAND=echo"

      # Colores para dircolors y LESS
      if command -v dircolors >/dev/null 2>&1; then
          test -r ~/.dircolors && eval "$(dircolors -b ~/.dircolors)" || eval "$(dircolors -b)"
          export LS_COLORS="$LS_COLORS:ow=30;44:"

          export LESS_TERMCAP_mb=$'\E[0m'
          export LESS_TERMCAP_md=$'\E[0m'
          export LESS_TERMCAP_me=$'\E[0m'
          export LESS_TERMCAP_so=$'\E[0m'
          export LESS_TERMCAP_se=$'\E[0m'
          export LESS_TERMCAP_us=$'\E[0m'
          export LESS_TERMCAP_ue=$'\E[0m'
      fi
      # Bienvenida
      fastfetch
      echo hallo bitch
    '';
  };
